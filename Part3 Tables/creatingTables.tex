	\section{Introduction to Tables in LaTeX}
	This document demonstrates various table creation techniques in \LaTeX, ranging from simple tabular environments to advanced features like multi-page tables, color styling, and cell merging.
	
	\section{Basic Table}
	A simple table without borders, using basic column alignment.
	
	\begin{center}
		\begin{tabular}{c c c}
			cell1 & cell2 & cell3 \\ 
			cell4 & cell5 & cell6 \\  
			cell7 & cell8 & cell9    
		\end{tabular}
	\end{center}
	
	\section{Table with Borders}
	Adding vertical borders improves readability and structure.
	
	\begin{center}
		\begin{tabular}{|c|c|c|}
			\hline
			cell1 & cell2 & cell3 \\ 
			\hline
			cell4 & cell5 & cell6 \\  
			\hline
			cell7 & cell8 & cell9 \\
			\hline
		\end{tabular}
	\end{center}

	\section{Multi-Page Table with Longtable}
	The \texttt{longtable} environment allows tables to span multiple pages, ideal for large datasets.
	
	\begin{longtable}[c]{|c|c|}
		\caption{Long Table Example}
		\label{long}\\
		
		\hline
		\multicolumn{2}{|c|}{\textbf{Beginning of Table}} \\
		\hline
		\textbf{Item} & \textbf{Value} \\
		\hline
		\endfirsthead
		
		\hline
		\multicolumn{2}{|c|}{\textbf{Continuation of Table~\ref{long}}} \\
		\hline
		\textbf{Item} & \textbf{Value} \\
		\hline
		\endhead
		
		\hline
		\multicolumn{2}{|r|}{\textit{Continued on next page}} \\
		\hline
		\endfoot

	\section*{Summary of Useful Commands}
	
	\begin{table}[h!]
		\centering
		\caption{Common Table Commands Reference}
		\label{table:commands}
		\begin{tabular}{|p{4cm}|p{8cm}|}
			\hline
			\textbf{Command} & \textbf{Description} \\
			\hline
			\texttt{\&} & Column separator \\
			\texttt{\textbackslash\textbackslash} & New row \\
			\texttt{\textbackslash hline} & Horizontal line across all columns \\
			\texttt{\textbackslash cline\{i-j\}} & Partial horizontal line from column i to j \\
			\texttt{\textbackslash multicolumn\{n\}\{fmt\}\{text\}} & Merge n columns with custom formatting \\
			\texttt{\textbackslash multirow\{n\}\{width\}\{text\}} & Merge n rows (requires multirow package) \\
			\texttt{\textbackslash rowcolor\{color\}} & Color entire row \\
			\texttt{\textbackslash cellcolor\{color\}} & Color individual cell \\
			\texttt{\textbackslash columncolor\{color\}} & Color entire column \\
			\texttt{\textbackslash toprule} & Top rule (booktabs) \\
			\texttt{\textbackslash midrule} & Middle rule (booktabs) \\
			\texttt{\textbackslash bottomrule} & Bottom rule (booktabs) \\
			\hline
		\end{tabular}
	\end{table}
