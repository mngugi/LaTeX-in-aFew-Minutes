\subsection*{Spaces}

\begin{align*}
	f(x) &= x^2\! +3x\! +2 \\
	f(x) &= x^2+3x+2 \\
	f(x) &= x^2\, +3x\, +2 \\
	f(x) &= x^2\: +3x\: +2 \\
	f(x) &= x^2\; +3x\; +2 \\
	f(x) &= x^2\ +3x\ +2 \\
	f(x) &= x^2\quad +3x\quad +2 \\
	f(x) &= x^2\qquad +3x\qquad +2
\end{align*}
