\subsection{Aligning Several Equations}

	2x - 5y &=  8 \\ 
	3x + 9y &=  -12
	
	x&=y           &  w &=z              &  a&=b+c\\
	2x&=-y         &  3w&=\frac{1}{2}z   &  a&=b\\
	-4 + 5x&=2+y   &  w+2&=-1+w          &  ab&=cb
\end{align*}
